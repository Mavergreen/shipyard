cmake_minimum_required(VERSION 3.16)
# LANGUAGES NONE: no C/CXX compiler enabled -> if including MavergreenCompatGuard
# fired RequireAppleClang, configure would FATAL. It must not.
project(standalone_include LANGUAGES NONE)
find_package(MavergreenShipyard REQUIRED)
include(MavergreenCompatGuard)
if(NOT COMMAND mavergreen_assert_binary_compatible)
  message(FATAL_ERROR "mavergreen_assert_binary_compatible not defined after include(MavergreenCompatGuard)")
endif()
include(MavergreenFetch)
if(NOT COMMAND mavergreen_fetch_sdk)
  message(FATAL_ERROR "mavergreen_fetch_sdk not defined after include(MavergreenFetch)")
endif()
include(MavericksSparkle)
if(NOT COMMAND mavericks_add_updater_app OR NOT COMMAND mavericks_fetch_sparkle)
  message(FATAL_ERROR "MavericksSparkle did not define its functions after include")
endif()
include(MavergreenSDKMetadata)
if(NOT COMMAND mavergreen_materialize_sdk_metadata)
  message(FATAL_ERROR "MavergreenSDKMetadata did not define mavergreen_materialize_sdk_metadata after include")
endif()
include(MavergreenHostTools)
if(NOT COMMAND mavergreen_build_host_tools)
  message(FATAL_ERROR "MavergreenHostTools did not define mavergreen_build_host_tools after include")
endif()
include(MavergreenXcodeInputs)
if(NOT COMMAND mavergreen_export_xcode_sources)
  message(FATAL_ERROR "MavergreenXcodeInputs did not define mavergreen_export_xcode_sources after include")
endif()
include(MavergreenResourceInputs)
if(NOT COMMAND mavergreen_extract_resource_members)
  message(FATAL_ERROR "MavergreenResourceInputs did not define mavergreen_extract_resource_members after include")
endif()
message(STATUS "standalone-include OK: helpers available without the AppleClang gate")
